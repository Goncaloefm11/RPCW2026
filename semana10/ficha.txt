SELECT DISTINCT ?cientista ?desc ?nome WHERE {
  ?cientista a dbo:Scientist .
  ?cientista dbo:description ?desc .
  ?cientista rdfs:label ?nome
   FILTER (lang(?nome) = "en")
   FILTER (lang(?desc) = "en")
}  LIMIT 20
