### 19. Qual a distribuição de aprendizes por disciplina? [cite: 169]
SELECT ?Disciplina (COUNT(?aprendiz) AS ?numAlunos) WHERE {
    ?aprendiz :aprende ?discIndividual .
    ?discIndividual :nome ?Disciplina .
}
GROUP BY ?Disciplina
ORDER BY DESC(?numAlunos)

### 20. Qual a distribuição de conceitos por período histórico? [cite: 170]
SELECT ?Período (COUNT(?conceito) AS ?numConceitos) WHERE {
    ?conceito :surgeEm ?perioIndividual .
    ?perioIndividual :nome ?Período .
}
GROUP BY ?Período
ORDER BY DESC(?numConceitos)

### 21. Quais os títulos das obras associadas a cada conceito? [cite: 172]
SELECT ?Conceito (GROUP_CONCAT(?obraTitulo; separator="; ") AS ?ListaDeObras) WHERE {
    ?obra :explica ?concIndividual .
    ?concIndividual :nome ?Conceito .
    ?obra :titulo ?obraTitulo .
}
GROUP BY ?Conceito

### 22. Quais os aprendizes que frequentam cada uma das disciplinas? [cite: 174]
SELECT ?Disciplina (GROUP_CONCAT(?aprendizNome; separator=", ") AS ?ListaDeAprendizes) WHERE {
    ?aprendiz :aprende ?discIndividual .
    ?aprendiz :nome ?aprendizNome .
    ?discIndividual :nome ?Disciplina .
}
GROUP BY ?Disciplina

### 23. Qual a distribuição de aprendizes por escalões etários? [cite: 179]
# Escalões: [17, 34], [35, 44], [45, 54], [55, 64], [65, 74], [75, 100] [cite: 181]
SELECT ?EscalãoEtário (COUNT(?aprendiz) AS ?numAprendizes) WHERE {
    ?aprendiz rdf:type :Aprendiz .
    ?aprendiz :idade ?idade .
    
    BIND(
        IF(?idade >= 17 && ?idade <= 34, "[17, 34]",
        IF(?idade >= 35 && ?idade <= 44, "[35, 44]",
        IF(?idade >= 45 && ?idade <= 54, "[45, 54]",
        IF(?idade >= 55 && ?idade <= 64, "[55, 64]",
        IF(?idade >= 65 && ?idade <= 74, "[65, 74]",
        IF(?idade >= 75 && ?idade <= 100, "[75, 100]", "Outro")))))) AS ?EscalãoEtário
    )
}
GROUP BY ?EscalãoEtário
ORDER BY ?EscalãoEtário

### 24. Query CONSTRUCT para a relação estudaCom [cite: 185]
# Lógica: Se um aluno aprende uma disciplina que um mestre ensina[cite: 46, 47].
