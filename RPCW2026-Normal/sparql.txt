# 17. Qual a distribuição de jogos por editora, ordenada descendentemente pela quantidade de jogos?

# 19. Quais os jogos com mecânicas mais complexas? (Jogos com mais de 3 mecânicas associadas)
# Nota: Usamos HAVING para filtrar agregações e GROUP_CONCAT para listar todas as mecânicas na mesma linha.
SELECT ?id ?nome (GROUP_CONCAT(?mechanicName; separator=", ") AS ?listaMecanicas) WHERE {
    ?mechanic :usedInGame ?game .
    ?mechanic :name ?mechanicName .
    ?game :id ?id .
    ?game :name ?nome .
}
GROUP BY ?id ?nome
HAVING (COUNT(?mechanic) > 3)
--------------------------------------------------------------------------------------

# 20. Planeamento do evento da ludoteca escolar: autores e mecânicas exclusivos da categoria "Family"
# As mecânicas aparecem agrupadas por jogo e autor, separadas por vírgulas.
SELECT ?nomeJogo ?nomeCriador (GROUP_CONCAT(?nomeMecanica; separator=", ") AS ?listaMecanicas) WHERE {
    ?game :category "Family" .
    ?game :name ?nomeJogo .
    
    # Cruzar com os autores
    ?author :designedGame ?game .
    ?author :name ?nomeCriador .
    
    # Cruzar com as mecânicas
    ?mechanic :usedInGame ?game .
    ?mechanic :name ?nomeMecanica .
}
GROUP BY ?nomeJogo ?nomeCriador
ORDER BY ?nomeJogo ?nomeCriador
--------------------------------------------------------------------------------------
